    string_out_loop:

    POP => ACC
    POP => ADR
    ACC => PUSH

    string_in_loop:
        IN => ACC
        ACC => LED
        
        NULL => MEM
        
        ' ' => CMP
        POP => PC z

        '\r' => CMP
        POP => PC z

        ACC => MEM

        ADR => ACC
        1 => ADD
        ACC => ADR
    string_in_loop => PC
